cmake_minimum_required(VERSION 3.25)
project(re_benchmark LANGUAGES C)
find_package(BEAM CONFIG REQUIRED COMPONENTS Runtime Host NIF)
option(RE_BENCH_CANDIDATE "Bind the independent package instead of legacy BIFs" OFF)
if(RE_BENCH_CANDIDATE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(BEAM_RE_TEST_INSTRUMENTATION ON CACHE BOOL "" FORCE)
    set(BEAM_RE_YIELD_COVERAGE OFF CACHE BOOL "" FORCE)
    set(BEAM_RE_NATIVE_KIND MODULE CACHE STRING "" FORCE)
    add_subdirectory("${PACKAGE_SOURCE}" native)
elseif(NOT BEAM_SDK_HAS_LEGACY_RE_BIFS)
    message(FATAL_ERROR "Baseline SDK must retain legacy re BIF registrations")
endif()
add_executable(re_benchmark main.c)
target_compile_features(re_benchmark PRIVATE c_std_11)
target_compile_definitions(re_benchmark PRIVATE _POSIX_C_SOURCE=200809L
    RE_BENCH_CANDIDATE=$<BOOL:${RE_BENCH_CANDIDATE}>)
target_include_directories(re_benchmark PRIVATE "${FIXTURE_DIRECTORY}")
target_link_libraries(re_benchmark PRIVATE BEAM::Runtime BEAM::Host)
set_target_properties(re_benchmark PROPERTIES ENABLE_EXPORTS TRUE)
